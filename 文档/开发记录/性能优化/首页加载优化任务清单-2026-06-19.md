     ```php
     public static function resolve(?string $value): ?string
     {
         if (!$value) return null;
         
         // 已是相对路径
         if (str_starts_with($value, '/uploads/')) {
             return $value;
         }
         
         // 包含完整域名，转换为相对路径
         if (str_contains($value, '://')) {
             $path = parse_url($value, PHP_URL_PATH);
             return $path ?: $value;
         }
         
         return '/uploads/' . ltrim($value, '/');
     }
     ```
